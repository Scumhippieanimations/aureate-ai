let mx=0;
let my=0;

let rx=0;
let ry=0;

const c=
document.getElementById(
"cursor"
);

const r=
document.getElementById(
"ring"
);


document.addEventListener(
"mousemove",
e=>{

mx=e.clientX;
my=e.clientY;

c.style.left=
mx+"px";

c.style.top=
my+"px";

}
);


function animate(){

rx+=(mx-rx)*0.12;
ry+=(my-ry)*0.12;

r.style.left=
rx+"px";

r.style.top=
ry+"px";

requestAnimationFrame(
animate
);

}

animate();

let images=
JSON.parse(
localStorage.getItem(
"aureateImages"
)
)||[];


function generateImage(){

const prompt=
document
.getElementById(
"imgPrompt"
)
.value;

const style=
document
.getElementById(
"imgStyle"
)
.value;


if(prompt==="") return;


const image={

prompt,
style,

id:Date.now()

};


images.unshift(
image
);

localStorage.setItem(
"aureateImages",
JSON.stringify(
images
)
);

loadGallery();

document
.getElementById(
"imgPrompt"
)
.value="";

}


function loadGallery(){

const gallery=
document
.getElementById(
"gallery"
);

gallery.innerHTML="";


images.forEach(
img=>{

gallery.innerHTML+=`

<div class="imageCard">

<div>

<b>${img.style}</b>

<br><br>

${img.prompt}

</div>

</div>

`;

});

}


loadGallery();
